cmake_minimum_required(VERSION 3.24)
project(Cords VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
find_package(Qt6 6.10.3 EXACT REQUIRED COMPONENTS Core Gui Network Qml Quick QuickControls2 Widgets QuickTest Test Concurrent)
qt_standard_project_setup(REQUIRES 6.10)
find_program(CARGO_EXECUTABLE cargo REQUIRED)

get_filename_component(CORDS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CORDS_CARGO_TARGET "${CMAKE_BINARY_DIR}/cargo" CACHE PATH "Cargo artifacts (outside source preferred)")
set(CORDS_CARGO_PROFILE "release" CACHE STRING "Cargo dev or release profile")
if(CORDS_CARGO_PROFILE STREQUAL "release")
    set(CORDS_CARGO_DIR release)
else()
    set(CORDS_CARGO_DIR debug)
endif()
if(WIN32)
    set(CORDS_RUST_LIB "${CORDS_CARGO_TARGET}/${CORDS_CARGO_DIR}/cords_qt_bridge.lib")
else()
    set(CORDS_RUST_LIB "${CORDS_CARGO_TARGET}/${CORDS_CARGO_DIR}/libcords_qt_bridge.a")
endif()
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "${CORDS_ROOT}/crates/*.rs" "${CMAKE_CURRENT_SOURCE_DIR}/rust/src/*.rs" "${CORDS_ROOT}/migrations/sqlite/*.sql")
add_custom_command(OUTPUT "${CORDS_RUST_LIB}"
    COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CORDS_CARGO_TARGET}" "CARGO_PROFILE_DEV_DEBUG=0"
        ${CARGO_EXECUTABLE} build --locked -p cords-qt-bridge --profile ${CORDS_CARGO_PROFILE}
    WORKING_DIRECTORY "${CORDS_ROOT}"
    DEPENDS ${RUST_SOURCES} rust/build.rs rust/Cargo.toml "${CORDS_ROOT}/Cargo.toml" "${CORDS_ROOT}/Cargo.lock"
    VERBATIM USES_TERMINAL)
add_custom_target(cords-rust DEPENDS "${CORDS_RUST_LIB}")
add_library(cords-bridge STATIC IMPORTED GLOBAL)
set_target_properties(cords-bridge PROPERTIES IMPORTED_LOCATION "${CORDS_RUST_LIB}")
add_dependencies(cords-bridge cords-rust)

qt_add_executable(Cords WIN32 MACOSX_BUNDLE cpp/main.cpp cpp/controller.cpp cpp/controller.h cpp/rows.h cpp/jxlimage.cpp cpp/jxlimage.h cpp/platformevents.cpp cpp/platformevents.h cpp/notifications.cpp cpp/notifications.h)
if(WIN32)
    target_sources(Cords PRIVATE assets/cords.rc)
elseif(APPLE)
    set_source_files_properties(assets/icon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_sources(Cords PRIVATE assets/icon.icns)
    set_target_properties(Cords PROPERTIES MACOSX_BUNDLE_ICON_FILE icon.icns)
endif()
set_target_properties(Cords PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "chat.cords.desktop"
    MACOSX_BUNDLE_BUNDLE_NAME "Cords" WIN32_EXECUTABLE TRUE)
target_include_directories(Cords PRIVATE "${CORDS_CARGO_TARGET}/cxxbridge" "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
option(CORDS_ALLOW_SOURCE_MIGRATIONS "Allow a development executable to read repository migrations" OFF)
if(CORDS_ALLOW_SOURCE_MIGRATIONS)
    target_compile_definitions(Cords PRIVATE CORDS_SOURCE_MIGRATIONS="${CORDS_ROOT}/migrations/sqlite")
endif()
target_link_libraries(Cords PRIVATE cords-bridge Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Concurrent)
if(MSVC)
    target_compile_options(Cords PRIVATE /W4 /utf-8 /permissive-)
    target_link_libraries(Cords PRIVATE advapi32 bcrypt crypt32 dbghelp kernel32 ncrypt ntdll ole32 oleaut32 secur32 shell32 user32 userenv ws2_32 runtimeobject wtsapi32)
else()
    target_compile_options(Cords PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(Cords PRIVATE ${CMAKE_DL_LIBS})
    if(UNIX AND NOT APPLE)
        target_link_libraries(Cords PRIVATE pthread m)
    endif()
endif()
file(GLOB QML_SOURCES CONFIGURE_DEPENDS "qml/*.qml")
foreach(source IN LISTS QML_SOURCES)
    get_filename_component(name "${source}" NAME)
    set_source_files_properties("${source}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
endforeach()
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(Cords URI Cords VERSION 1.0 QML_FILES ${QML_SOURCES})
qt_add_resources(Cords "cords-icon" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}/assets"
    FILES "${CMAKE_CURRENT_SOURCE_DIR}/assets/128x128.png")

include(GNUInstallDirs)
install(TARGETS Cords BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(APPLE)
    set(CORDS_DATA_DESTINATION "Cords.app/Contents/Resources/cords")
else()
    set(CORDS_DATA_DESTINATION "${CMAKE_INSTALL_DATADIR}/cords")
endif()
install(DIRECTORY "${CORDS_ROOT}/migrations/sqlite" DESTINATION "${CORDS_DATA_DESTINATION}/migrations")
install(FILES "${CORDS_ROOT}/LICENSE" assets/DISTRIBUTION.md DESTINATION "${CORDS_DATA_DESTINATION}")
install(DIRECTORY "${CORDS_ROOT}/assets/avatar-codec/licenses" DESTINATION "${CORDS_DATA_DESTINATION}/avatar-codec")
if(UNIX AND NOT APPLE)
    install(FILES assets/chat.cords.desktop.desktop DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
    install(FILES assets/128x128.png DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/128x128/apps" RENAME chat.cords.desktop.png)
endif()
qt_generate_deploy_qml_app_script(TARGET Cords OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})

enable_testing()
add_test(NAME qml-controls COMMAND Qt6::qmltestrunner -input "${CMAKE_CURRENT_SOURCE_DIR}/tests" -import "${CMAKE_CURRENT_SOURCE_DIR}/qml")
set_tests_properties(qml-controls PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")

qt_add_executable(CordsMediaTests tests/tst_media.cpp cpp/jxlimage.cpp cpp/jxlimage.h)
target_include_directories(CordsMediaTests PRIVATE "${CORDS_CARGO_TARGET}/cxxbridge" "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
get_target_property(CORDS_NATIVE_LIBRARIES Cords LINK_LIBRARIES)
target_link_libraries(CordsMediaTests PRIVATE ${CORDS_NATIVE_LIBRARIES} Qt6::Test)
target_compile_definitions(CordsMediaTests PRIVATE
    CORDS_TEST_IMAGE="${CORDS_ROOT}/crates/cords-avatar-codec/tests/fixtures/looping-a.jxl"
    CORDS_TEST_CAPTURE="${CMAKE_BINARY_DIR}/native-media.png")
if(MSVC)
    target_compile_options(CordsMediaTests PRIVATE /utf-8)
endif()
add_test(NAME native-media COMMAND CordsMediaTests -o "${CMAKE_BINARY_DIR}/native-media-tests.txt,txt")
set_tests_properties(native-media PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 120)

qt_add_executable(CordsControllerTests tests/tst_controller.cpp cpp/controller.cpp cpp/controller.h cpp/rows.h cpp/platformevents.cpp cpp/platformevents.h)
target_include_directories(CordsControllerTests PRIVATE "${CORDS_CARGO_TARGET}/cxxbridge" "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
target_link_libraries(CordsControllerTests PRIVATE ${CORDS_NATIVE_LIBRARIES} Qt6::Test Qt6::Network)
target_compile_definitions(CordsControllerTests PRIVATE CORDS_TEST_MIGRATIONS="${CORDS_ROOT}/migrations/sqlite")
if(MSVC)
    target_compile_options(CordsControllerTests PRIVATE /utf-8)
endif()
add_test(NAME native-controller COMMAND CordsControllerTests)
set_tests_properties(native-controller PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 45)
